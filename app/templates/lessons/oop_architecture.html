01. Master Blueprint to Object-Oriented Programming (OOP) in Python

PART 1: Classes, Object Instantiation, and the Dunder Methods

A Class acts as a structural template. An Object is a living copy of that template.

class EmptyTemplate:
    # 'pass' prevents an empty block syntax indentation crash
    pass


class Person:
    # Class Property: Shared globally across every instance generated
    species = "Human"

    # The __init__ method is a special 'dunder' (double underscore) initializer.
    # It executes automatically whenever a brand new instance is being built.
    # 'self' represents the unique address pointer of the object currently being created.
    def __init__(self, name: str, age: int = 18):
        # Instance Properties: Unique, separate values bound to this specific object
        self.name = name
        self.age = age

    # Magic/Dunder method controlling how an object looks when converted into a string
    def __str__(self):
        return f"Person (Name: {self.name}, Age: {self.age})"

    # Instance Method: Custom behavior that reads or modifies internal state via 'self'
    def celebrate_birthday(self):
        self.age += 1
        print(f"Telemetry Event: {self.name} just aged up to {self.age}.")


# --- Verification Pipeline ---
print("--- 1. Objects & In-Place Modification ---")
p1 = Person("Emil", 35)
p2 = Person("Tobias", 24)

print(p1)  # Triggers __str__ automatically
print(f"Direct property access: {p1.name} belongs to species: {p1.species}")

# Modifying instance properties directly on the fly
p2.age = 25
p2.celebrate_birthday()

# Dynamically injecting specific unique tracking assets into singular instances only
p1.assigned_sector = "Main Lab"
print(f"Instance unique attribute successfully mapped: {p1.assigned_sector}")

# Cleanup operations: Purging attributes or whole objects from active tracking memory
del p2.age  # 'p2.age' is now deleted; accessing it would cause an AttributeError
del p2      # 'p2' object completely destroyed from runtime stack

PART 2: The Four Pillars - Inheritance & Polymorphism

  • Inheritance: Passing parental methods down into specialized child structures.
  • Polymorphism: Different objects executing identically named methods in their own way.
class Vehicle:
    def __init__(self, brand: str, model: str):
        self.brand = brand
        self.model = model

    def move(self):
        print("System Alert: General generic vehicle propulsion activated.")


# Child class inherits everything from parent Vehicle
class Car(Vehicle):
    # If no __init__ is defined, it inherits the parent's constructor automatically.
    def move(self):
        # Polymorphism via Method Overriding: Changing default parent logic
        print(f"Drive! The {self.brand} {self.model} is racing down the highway.")


class Boat(Vehicle):
    def move(self):
        print(f"Sail! The {self.brand} {self.model} is slicing through ocean waves.")


class Plane(Vehicle):
    def __init__(self, brand: str, model: str, maximum_altitude: int):
        # super() references the parent class and safely runs its initializer
        super().__init__(brand, model)
        self.maximum_altitude = maximum_altitude

    def move(self):
        print(f"Fly! The {self.brand} {self.model} climbed to {self.maximum_altitude} feet.")


# --- Verification Pipeline ---
print("\n--- 2. Inheritance & Polymorphic Dispatch ---")
fleet = [
    Car("Ford", "Mustang"),
    Boat("Ibiza", "Touring 20"),
    Plane("Boeing", "747", 35000)
]

# Polymorphism allows us to execute the same method across entirely different types seamlessly
for vehicle in fleet:
    print(f"Unit Metadata: {vehicle.brand} {vehicle.model}")
    vehicle.move()  # Each object responds uniquely using its own version of the method

PART 3: The Four Pillars - Encapsulation & Data Accessibility

Encapsulation safeguards internal variables from rogue external modifications.

  • _prefix → Protected attribute (Convention only, please do not alter externally)
  • __prefix → Private attribute (Triggers Name Mangling to strictly lock it out)
class SecurityVault:
    def __init__(self, owner: str, initial_funds: float):
        self.owner = owner          # Public
        self._clearance_level = 2   # Protected
        self.__balance = initial_funds  # Private (Mangled to _SecurityVault__balance)

    # Getter Method: Controlled, read-only extraction interface
    def get_balance(self) -> float:
        return self.__balance

    # Setter Method: Sanitizes and filters values before modifying internal memory
    def set_balance(self, new_amount: float):
        if new_amount >= 0:
            self.__balance = new_amount
            print(f"Vault Notification: Ledger balance updated to {self.__balance}.")
        else:
            print("[SECURITY DENIED] Negative cash allocation rejected.")


# --- Verification Pipeline ---
print("\n--- 3. Data Protection & Encapsulation Rules ---")
vault = SecurityVault("Operator-01", 50000.0)

# print(vault.__balance) # CRASHES! AttributeError: 'SecurityVault' object has no attribute '__balance'
print(f"Authorized extraction via Getter Interface: {vault.get_balance()}")
vault.set_balance(65000.0)

# Demonstration of Name Mangling structural conversion mechanics
print(f"Mangled backdoor override value: {vault._SecurityVault__balance}")

PART 4: Inner (Nested) Classes

An Inner Class is declared inside a parent Outer Class to model closely bound helper structures.

class Computer:
    def __init__(self, cpu_model: str, ram_capacity: int):
        # Automatically instantiating internal sub-components
        self.cpu = self.CPU(cpu_model)
        self.ram = self.RAM(ram_capacity)

    def display_system_profile(self):
        print(f"Computer Host Matrix:")
        self.cpu.process()
        self.ram.store()

    # Inner Class Alpha
    class CPU:
        def __init__(self, model: str):
            self.model = model

        def process(self):
            print(f" -> CPU [{self.model}] calculation cores active.")

    # Inner Class Beta
    class RAM:
        def __init__(self, capacity: int):
            self.capacity = capacity

        def store(self):
            print(f" -> RAM [{self.capacity}GB] memory grids loaded.")


# --- Verification Pipeline ---
print("\n--- 4. Nested Structural Architectures ---")
workstation = Computer("Intel i9", 64)
workstation.display_system_profile()
